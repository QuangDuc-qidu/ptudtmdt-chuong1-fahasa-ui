8.9 Functions as Objects

Tham số tax giữ một function. calculateTax gọi tax() để lấy tỷ lệ. Hàm vô danh được truyền ngay khi gọi, nên có thể thay quy tắc mà không viết lại phép nhân. toFixed(2) trả chuỗi với hai chữ số thập phân. Ví dụ thuế theo đúng giáo trình, không phải quy tắc thuế của cửa hàng nhóm.

Đối chiếu file: Lab08/exercise09.html và Lab08/js/exercise09.js.

var isCanadian = true;
function calculateTax(amount, tax) { return amount * tax(); }
function calculateTotal(price, quantity) {
  var amount = price * quantity;
  return amount + calculateTax(amount, function () { return isCanadian ? 0.05 : 0.0; });
}
document.write('<p>Price = $15.00; quantity = 2; thuế = 5%</p>');
document.write('<p id="tax-total">Total = $' + calculateTotal(15, 2).toFixed(2) + '</p>');
isCanadian = false;
document.write('<p>Không áp dụng thuế: $' + calculateTotal(15, 2).toFixed(2) + '</p>');
isCanadian = true;

Tự thử: Kiểm tra 15 × 2 = 30; cộng thuế 5% thành 31.50, không thuế là 30.00.

function Order(product, price, quantity) {
  this.product = product; this.price = price; this.quantity = quantity;
  this.total = function () { return this.price * this.quantity; };

countries.push(new Country('Spain', 'ES', 'Madrid', 46505963));

var btn = document.getElementById('testButton');
var content = document.getElementById('content');
var pendingTimer;
function simpleHandler(event) {
  clearTimeout(pendingTimer);
  if (btn.textContent === 'Hide') {
    btn.textContent = 'Show'; content.className = 'makeItDisappear';
    pendingTimer = setTimeout(function () { content.style.display = 'none'; }, 1000);
  } else {
    btn.textContent = 'Hide'; content.style.display = 'block';
    pendingTimer = setTimeout(function () { content.className = 'makeItNormal'; }, 50);
  }
}
btn.addEventListener('click', simpleHandler);
var img = document.getElementById('mainImage');
img.addEventListener('mouseover', function () { img.className = 'makeItGray'; });
img.addEventListener('mouseout', function () { img.className = 'makeItNormal'; });

window.addEventListener('load', function () {
  var divToGet = document.getElementById('div1');
